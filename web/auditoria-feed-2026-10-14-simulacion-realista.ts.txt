/* ---------- PROTOTIPO v2 (solo medición; no es código del proyecto) ---------- */
function v2(snap: FeedSnapshot, cfg: FeedConfig, seed: string, round: number, o: Opts) {
  const sorted = [...snap.cases].sort((a, b) => (a.contentId < b.contentId ? -1 : 1))
  const picked = new Map(sorted.map((c) => [c.contentId, rotateCasePins(seed, round, c)]))
  const casePool: GeneratedPin[] = []
  if (o.caseMode === 'layers') {
    const maxF = Math.max(0, ...sorted.map((c) => c.force))
    let prev: string | null = null
    for (let l = 0; l < maxF; l++) {
      const layer = shuffled(sorted.filter((c) => c.force > l), deriveRng(seed, 'layer', round, l))
      if (layer.length > 1 && layer[0].contentId === prev) [layer[0], layer[1]] = [layer[1], layer[0]]
      for (const c of layer) casePool.push({ pinId: picked.get(c.contentId)![l].pinId, contentId: c.contentId, kind: 'case', relaxationLevel: 0 })
      prev = layer.length ? layer[layer.length - 1].contentId : prev
    }
  } else {
    const rng = deriveRng(seed, 'strat', round)
    const keyed: { key: number; pin: GeneratedPin }[] = []
    for (const c of sorted) {
      const u = rng()
      picked.get(c.contentId)!.forEach((rp, k) => keyed.push({ key: (k + u) / c.force, pin: { pinId: rp.pinId, contentId: c.contentId, kind: 'case', relaxationLevel: 0 } }))
    }
    keyed.sort((a, b) => a.key - b.key)
    keyed.forEach((k) => casePool.push(k.pin))
  }
  const nCase = casePool.length
  const nonCase = snap.insights.length + snap.tools.length + snap.channel.length + snap.other.length
  const T = nCase > 0 ? Math.ceil(nCase / (cfg.ratios.cases / 100)) : nonCase
  const avail = { ...cfg.ratios, insights: snap.insights.length ? cfg.ratios.insights : 0, tools: snap.tools.length ? cfg.ratios.tools : 0, channel: snap.channel.length ? cfg.ratios.channel : 0, other: snap.other.length ? cfg.ratios.other : 0 }
  const quota = largestRemainderQuotas(T - nCase, avail)
  const q: Record<string, number> = { case: nCase, insight: quota.insights, tool: quota.tools, channel: quota.channel, other: quota.other }
  const queues: Record<string, GeneratedPin[]> = {
    case: casePool,
    insight: circularFillPool(seed, round, 'insight', snap.insights, quota.insights),
    tool: circularFillPool(seed, round, 'tool', snap.tools, quota.tools),
    channel: circularFillPool(seed, round, 'channel', snap.channel, quota.channel),
    other: circularFillPool(seed, round, 'other', snap.other, quota.other),
  }
  const total = Object.values(q).reduce((a, b) => a + b, 0)
  const rng = deriveRng(seed, 'skeleton', round)
  const ks = Object.keys(q).filter((k) => q[k] > 0)
  const placed: Record<string, number> = {}
  ks.forEach((k) => (placed[k] = 0))
  const sk: string[] = []
  for (let t = 0; t < total; t++) {
    const last = sk[t - 1]
    const cands = ks.filter((k) => placed[k] < q[k]).map((k) => ({ k, s: (q[k] * (t + 1)) / total - placed[k] + rng() * 0.05 })).sort((a, b) => b.s - a.s)
    const ok = (k: string) => k === 'case' || (o.strict ? last === 'case' || last === undefined : last !== k)
    sk.push((cands.find((x) => ok(x.k)) ?? cands[0]).k)
    placed[sk[t]]++
  }
  const idx: Record<string, number> = {}
  const seq = sk.map((k) => { idx[k] = (idx[k] ?? 0) + 1; return queues[k][idx[k] - 1] })
  return { sequence: seq, quota: q, T: total }
}

/* ---------- Catálogos ---------- */
const mk = (p: string, n: number, pins: number) => Array.from({ length: n }, (_, i) => ({ contentId: `${p}${String(i).padStart(3, '0')}`, pinIds: Array.from({ length: pins }, (_, j) => `${p}${i}-${j}`) }))
function forcesFor(n: number, dist: number[]): number[] {
  // dist = pesos para fuerza 1..5; reparto determinista
  const sum = dist.reduce((a, b) => a + b, 0); const out: number[] = []
  dist.forEach((w, i) => { for (let k = 0; k < Math.round((w / sum) * n); k++) out.push(i + 1) })
  while (out.length < n) out.push(1); return out.slice(0, n)
}
function catalog(nCases: number, dist: number[], t: number, i: number, e: number, o: number): FeedSnapshot {
  const fs = forcesFor(nCases, dist)
  return { cases: mk('c', nCases, 8).map((c, n) => ({ ...c, force: fs[n] })), insights: mk('i', i, 3), tools: mk('t', t, 4), channel: mk('h', e, 4), other: mk('o', o, 2) }
}

function analyse(seq: GeneratedPin[], snap: FeedSnapshot) {
  const force = new Map(snap.cases.map((c) => [c.contentId, c.force]))
  const nCases = snap.cases.length
  let maxRunCase = 0, run = 0, smallSameAdj = 0, smallAnyAdj = 0
  for (let i = 0; i < seq.length; i++) {
    if (seq[i].kind === 'case') { run++; maxRunCase = Math.max(maxRunCase, run) } else run = 0
    if (i > 0 && seq[i].kind !== 'case' && seq[i - 1].kind !== 'case') { smallAnyAdj++; if (seq[i].kind === seq[i - 1].kind) smallSameAdj++ }
  }
  const exposure: Record<number, { seen: number[]; app: number[]; n: number[] }> = {}
  const Ks = [40, 100, 200, 400]
  const byK: Record<number, number> = {}
  for (const K of Ks) {
    const cnt = new Map<string, number>()
    for (const p of seq.slice(0, K)) if (p.kind === 'case') cnt.set(p.contentId, (cnt.get(p.contentId) ?? 0) + 1)
    byK[K] = cnt.size / nCases
    exposure[K] = { seen: [0, 0, 0, 0, 0, 0], app: [0, 0, 0, 0, 0, 0], n: [0, 0, 0, 0, 0, 0] }
    for (const c of snap.cases) { const f = c.force; exposure[K].n[f]++; if (cnt.has(c.contentId)) exposure[K].seen[f]++; exposure[K].app[f] += cnt.get(c.contentId) ?? 0 }
  }
  // repetición de casos antes de haber visto todos
  const seen = new Set<string>(); let repeatsBeforeAll = 0; let allSeenAt = -1; let minGapSum = 0, gapN = 0
  const lastPos = new Map<string, number>()
  seq.forEach((p, i) => {
    if (p.kind !== 'case') return
    if (seen.has(p.contentId)) { if (seen.size < nCases) repeatsBeforeAll++; const g = i - lastPos.get(p.contentId)!; minGapSum += g; gapN++ } else seen.add(p.contentId)
    if (seen.size === nCases && allSeenAt < 0) allSeenAt = i
    lastPos.set(p.contentId, i)
  })
  // repetición de no-casos antes de agotar su tipo (en la ronda)
  const kinds: Record<string, { total: number; set: Set<string>; repBefore: number }> = {
    insight: { total: snap.insights.length, set: new Set(), repBefore: 0 }, tool: { total: snap.tools.length, set: new Set(), repBefore: 0 },
    channel: { total: snap.channel.length, set: new Set(), repBefore: 0 }, other: { total: snap.other.length, set: new Set(), repBefore: 0 } }
  for (const p of seq) { const k = kinds[p.kind]; if (!k) continue; if (k.set.has(p.contentId)) { if (k.set.size < k.total) k.repBefore++ } else k.set.add(p.contentId) }
  return { maxRunCase, smallSameAdj, smallAnyAdj, byK, exposure, repeatsBeforeAll, allSeenAt, meanGap: gapN ? minGapSum / gapN : 0, nonCaseRepBefore: Object.values(kinds).reduce((s, k) => s + k.repBefore, 0), force }
}
